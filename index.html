Integrasi Otomatis Google Spreadsheet

Pastikan kode di Google Apps Script sudah diperbarui dengan script di bawah ini (pilih Terapkan → Kelola deployment → Edit → Versi Baru) agar kolom S1–S30 dan tab Jawaban Per Nomor tampil di Sheet.

Kode Script Google Sheets (Rekap S1–S30 + Tab Jawaban Per Nomor): 
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheets()[0];
  var data = JSON.parse(e.postData.contents);
  var details = data.detailJawaban || [];
  var kunci = data.kunciJawaban || [];
  var totalSoal = data.total || details.length || 30;

  var baseHeaders = ["Waktu", "Nama", "Kelas", "NIS", "Nilai", "Benar", "Total Soal", "Indikasi Curang"];
  var soalHeaders = [];
  for (var i = 1; i <= totalSoal; i++) {
    soalHeaders.push("S" + i);
  }

  if (sheet.getLastRow() === 0) {
    sheet.appendRow(baseHeaders.concat(soalHeaders));
  } else if (sheet.getLastColumn() < baseHeaders.length + totalSoal) {
    sheet.getRange(1, baseHeaders.length + 1, 1, totalSoal).setValues([soalHeaders]).setFontWeight("bold");
  }

  sheet.appendRow([
    data.waktu, data.nama, data.kelas, data.nis,
    data.nilai, data.benar, totalSoal, data.curang
  ].concat(details));

  var detailSheet = ss.getSheetByName("Jawaban Per Nomor") || ss.insertSheet("Jawaban Per Nomor");
  if (detailSheet.getLastRow() === 0) {
    var dHeaders = ["Waktu", "Nama", "Kelas", "NIS", "Nilai", "Benar"];
    for (var j = 1; j <= totalSoal; j++) dHeaders.push("No. " + j);
    detailSheet.appendRow(dHeaders);
    if (kunci.length > 0) {
      detailSheet.appendRow(["-", "KUNCI JAWABAN", "-", "-", 100, totalSoal].concat(kunci));
    }
  }
  detailSheet.appendRow([data.waktu, data.nama, data.kelas, data.nis, data.nilai, data.benar].concat(details));
  return ContentService.createTextOutput("Sukses");
}
